z_fraction_proxy = (z-z_min)/(z_max-z_min)
theta_fraction_proxy = (theta-theta_min)/(theta_max-theta_min)
nearest_*_boundary_fraction_proxy = min(fraction, 1-fraction)
signed_*_position_proxy = 2*fraction-1
